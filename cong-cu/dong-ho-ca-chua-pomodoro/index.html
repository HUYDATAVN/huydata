    g.gain.linearRampToValueAtTime(vol, ctx.currentTime+0.6);

    const nodes = {src, g, extra:[]};


    if(type==="brown"){

      const lp = ctx.createBiquadFilter(); lp.type="lowpass"; lp.frequency.value=1000;

      src.connect(lp); lp.connect(g); nodes.extra.push(lp);

    } else { // white

      src.connect(g);

    }

    g.connect(ctx.destination);

    src.start();

    ambientNodes = nodes;

  }

  function stopAmbient(){

    if(!ambientNodes || !actx) { ambientNodes=null; return; }

    const ctx = actx, n = ambientNodes; ambientNodes = null;

    try{

      const now = ctx.currentTime;

      n.g.gain.cancelScheduledValues(now);

      n.g.gain.setValueAtTime(Math.max(0.0001,n.g.gain.value), now);

      n.g.gain.linearRampToValueAtTime(0.0001, now+0.3);

      n.src.stop(now+0.36);

      n.extra.forEach(x=>{ if(x && x.stop){ try{x.stop(now+0.36);}catch(e){} } });

    }catch(e){}

  }

    else stopAmbient();

  }

  function durMs(m){ return settings[m]*60000; }


  function setMode(m, silent){

    mode = m;

    document.body.className = "mode-"+m;

  function start(){

    if(running) return;

    audio();

    running = true;

    targetTime = Date.now() + remainingMs;

    lastWholeSecond = Math.ceil(remainingMs/1000);

    $("#startLabel").textContent = "Tạm dừng";

    $("#startIcon").innerHTML = '<rect x="6" y="5" width="4" height="14" rx="1"></rect><rect x="14" y="5" width="4" height="14" rx="1"></rect>';

    loopId = setInterval(loop, 100);

    refreshAmbient();

    updateAutoHint();

  }

  function pause(){

    if(!running) return;

    running = false;

    remainingMs = Math.max(0, targetTime - Date.now());

    clearInterval(loopId); loopId=null;

    $("#startLabel").textContent = "Tiếp tục";

    $("#startIcon").innerHTML = '<path d="M8 5v14l11-7L8 5Z"></path>';

    stopAmbient(); updateAutoHint();

  }

  function stop(){

    running=false; clearInterval(loopId); loopId=null;

    $("#startLabel").textContent = "Bắt đầu";

    $("#startIcon").innerHTML = '<path d="M8 5v14l11-7L8 5Z"></path>';

    stopAmbient(); updateAutoHint();

  }

  function toggle(){ running ? pause() : start(); }

  function reset(){ stop(); remainingMs = durMs(mode); lastWholeSecond=null; render(); }


  function loop(){

    remainingMs = targetTime - Date.now();

    if(remainingMs <= 0){ remainingMs=0; render(); complete(); return; }

    const whole = Math.ceil(remainingMs/1000);

    if(whole !== lastWholeSecond){ lastWholeSecond = whole; playTick(whole%2===0); }

    render();

  }


  function complete(){

    stop();

    if(settings.alarm) playAlarm();

    let next;

    if(mode==="focus"){

      completedFocus++;

      const t = tasks.find(x=>x.id===activeTaskId && !x.done);

      if(t){ t.done_pomos = (t.done_pomos||0)+1; if((t.done_pomos>=t.est)) {/* để người dùng tự tick xong */} renderTasks(); }

      const isLong = completedFocus % settings.interval === 0;

      next = isLong ? "long" : "short";

      notify("Hết phiên tập trung 🎉", isLong?"Nghỉ dài thôi, bạn xứng đáng!":"Nghỉ ngắn một chút nhé.");

    } else {

      next = "focus";

      notify("Hết giờ nghỉ ⏱️", "Quay lại tập trung nào!");

    }

    persist();

    setMode(next, true);

    if(settings.auto){ start(); }

    else { updateAutoHint(); }

  }


  function skip(){

    const next = order[(order.indexOf(mode)+1) % order.length];

    setMode(next);

  }
